\documentclass[aspectratio=169,10pt]{beamer}
\usetheme{Madrid}

% Tắt shadow để giao diện phẳng và tránh lỗi vệt đen
\setbeamertemplate{blocks}[rounded][shadow=false]
\setbeamertemplate{title page}[default][colsep=-4bp,rounded=true,shadow=false]
\setbeamertemplate{navigation symbols}{}

% Refined color palette
\definecolor{primaryblue}{RGB}{20, 55, 110}
\setbeamercolor{palette primary}{bg=primaryblue,fg=white}
\setbeamercolor{structure}{fg=primaryblue}

\begin{document}

\begin{frame}{Geometric Examples}
    \begin{itemize}
        \item \textbf{Area of a rectangle}:
        \[
            A = xy
        \]
        ($x$ is length, $y$ is width).
        
        \vspace{0.4cm}
        
        \item \textbf{Volume of a circular cone}:
        \[
            V = \frac{1}{3}\pi r^2 h
        \]
        ($r$ is radius, $h$ is height).
    \end{itemize}
\end{frame}

\end{document}
